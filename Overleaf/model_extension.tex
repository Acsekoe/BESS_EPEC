\section{Scope}

The model is a deterministic equilibrium problem with equilibrium constraints
(EPEC) for battery energy storage system (BESS) capacity investment in a nodal
electricity market. Each investor chooses nodal power and energy capacity.
Conditional on those capacities, an independent system operator (ISO)
minimizes the physical cost of generation and BESS operation.

The formulation uses fixed demand, exact nodal and system balances, and an
exact primal--dual representation of the lower-level linear program. It
contains no strategic operating bids, load shedding, dispatch regularization,
leader-side price penalty, proximal term, Tikhonov term, minimum-norm price
selector, or relaxed complementarity condition. During an investor's best
response, rival capacities are fixed parameters; the capacity profile is
updated outside the mathematical program by diagonalization.

Unless indicated otherwise, sums extend over the full set associated with each
index. Power, energy, and monetary quantities are measured in MW, MWh, and EUR,
respectively.

\section{Investor problem}

Let $X_i=(X_i^P,X_i^E)$ denote investor $i$'s capacity decision and let
$X_{-i}$ collect the fixed capacities of its rivals. Feasible investment
decisions satisfy
\begin{subequations}\label{eq:investment}
\begin{align}
0
&\leq X^P_{i,n}
\leq \overline X_n-\sum_{\substack{j\in\mathcal I\\j\neq i}}X^P_{j,n},
&& \forall n\in\mathcal N,
\label{eq:headroom}\\
r^{\min}X^P_{i,n}
&\leq X^E_{i,n}
\leq r^{\max}X^P_{i,n},
&& \forall n\in\mathcal N.
\label{eq:duration}
\end{align}
\end{subequations}
The capital-recovery factor and daily annualized investment cost are
\begin{subequations}\label{eq:capital-cost}
\begin{align}
\operatorname{CRF}_i
&=
\begin{cases}
\displaystyle
\frac{w_i(1+w_i)^{L_i}}{(1+w_i)^{L_i}-1},
& w_i>0,\\[1.1ex]
\displaystyle \frac{1}{L_i},
& w_i=0,
\end{cases}
\label{eq:crf}\\
K_i
&=\frac{\operatorname{CRF}_i}{365.25}
\sum_{n\in\mathcal N}
\left(c_i^P X^P_{i,n}+c_i^E X^E_{i,n}\right).
\label{eq:daily-capital-cost}
\end{align}
\end{subequations}

Investor $i$ maximizes the daily profit
\begin{subequations}\label{eq:profit}
\begin{align}
\Pi_i
&=R_i^{\mathrm{BESS}}+R_i^{\mathrm{gen}}
-C_i^{\mathrm{deg}}-K_i,
\label{eq:profit-total}\\
R_i^{\mathrm{BESS}}
&=\sum_{n\in\mathcal N}\sum_{t\in\mathcal T}
\lambda_{n,t}
\left(p^{\mathrm{dis}}_{i,n,t}-p^{\mathrm{ch}}_{i,n,t}\right),
\label{eq:storage-revenue}\\
R_i^{\mathrm{gen}}
&=\sum_{g\in\mathcal G}\sum_{t\in\mathcal T}
s_{i,g}\left(\lambda_{n(g),t}-C_g\right)P_{g,t},
\label{eq:generation-rent}\\
C_i^{\mathrm{deg}}
&=\frac{d_i}{2}
\sum_{n\in\mathcal N}\sum_{t\in\mathcal T}
\left(p^{\mathrm{ch}}_{i,n,t}+p^{\mathrm{dis}}_{i,n,t}\right).
\label{eq:degradation-cost}
\end{align}
\end{subequations}
For a stand-alone merchant, $s_{i,g}=0$ for every $g$. A portfolio investor
additionally internalizes the spot-market rent of its owned generation shares.

\section{Lower-level market clearing}

For fixed capacity profile $X=(X_i,X_{-i})$, the ISO solves
\begin{equation}\label{eq:llp-objective}
\begin{aligned}
\underset{P,p^{\mathrm{ch}},p^{\mathrm{dis}},e,z}{\operatorname{minimize}}
\quad
Z^P
:={}&
\sum_{g\in\mathcal G}\sum_{t\in\mathcal T}C_gP_{g,t}\\
&+\frac{1}{2}
\sum_{j\in\mathcal I}\sum_{n\in\mathcal N}\sum_{t\in\mathcal T}
d_j\left(p^{\mathrm{ch}}_{j,n,t}+p^{\mathrm{dis}}_{j,n,t}\right)
\end{aligned}
\end{equation}
Define the line-flow expression
$f_{l,t}:=\sum_{n\in\mathcal N}H_{l,n}z_{n,t}$. The primal-feasibility
conditions are
\begin{subequations}\label{eq:llp-primal}
\begin{align}
z_{n,t}
&=
\sum_{g\in\mathcal G_n}P_{g,t}
+\sum_{j\in\mathcal I}
\left(p^{\mathrm{dis}}_{j,n,t}-p^{\mathrm{ch}}_{j,n,t}\right)
-D_{n,t},
&& \forall n\in\mathcal N,\ t\in\mathcal T,
\label{eq:nodal-balance}\\
\sum_{n\in\mathcal N}z_{n,t}
&=0,
&& \forall t\in\mathcal T,
\label{eq:system-balance}\\
0\leq P_{g,t}
&\leq\overline P_{g,t},
&& \forall g\in\mathcal G,\ t\in\mathcal T,
\label{eq:generation-bounds}\\
-\overline F_l
\leq f_{l,t}
&\leq\overline F_l,
&& \forall l\in\mathcal L,\ t\in\mathcal T,
\label{eq:line-bounds}\\
0\leq p^{\mathrm{ch}}_{j,n,t}
&\leq X^P_{j,n},
&& \forall j\in\mathcal I,\ n\in\mathcal N,\ t\in\mathcal T,
\label{eq:charge-bounds}\\
0\leq p^{\mathrm{dis}}_{j,n,t}
&\leq X^P_{j,n},
&& \forall j\in\mathcal I,\ n\in\mathcal N,\ t\in\mathcal T,
\label{eq:discharge-bounds}\\
e_{j,n,t}
&=e_{j,n,t-1}
+\eta p^{\mathrm{ch}}_{j,n,t}
-\eta^{-1}p^{\mathrm{dis}}_{j,n,t},
&& \forall j\in\mathcal I,\ n\in\mathcal N,\ t\in\mathcal T,
\label{eq:soc-transition}\\
0\leq e_{j,n,\tau}
&\leq X^E_{j,n},
&& \forall j\in\mathcal I,\ n\in\mathcal N,\ \tau\in\mathcal T_0,
\label{eq:soc-bounds}\\
e_{j,n,0}
&=e_{j,n,T},
&& \forall j\in\mathcal I,\ n\in\mathcal N.
\label{eq:soc-periodicity}
\end{align}
\end{subequations}
Equation~\eqref{eq:soc-periodicity} imposes a periodic daily cycle; the model
does not prescribe a fixed initial state of charge.

\section{Exact lower-level optimality}

The lower-level linear program is embedded through primal feasibility, dual
feasibility, and strong duality. The dual-variable domains follow the
minimization convention used by the implementation:
\begin{align}
\nu_{g,t},\ \mu^+_{l,t},\ \rho_{j,n,t},\
\sigma_{j,n,t},\ \delta_{j,n,\tau}
&\leq 0,
\label{eq:nonpositive-duals}\\
\mu^-_{l,t}
&\geq 0,
\label{eq:nonnegative-duals}\\
\lambda_{n,t},\ \lambda_t^{\mathrm{sys}},\
\gamma_{j,n,t},\ \rho^{\mathrm{per}}_{j,n}
&\in\mathbb R.
\label{eq:free-duals}
\end{align}
Stationarity with respect to generation, charging, discharging, and net
injection gives
\begin{subequations}\label{eq:llp-dual}
\begin{align}
\lambda_{n(g),t}+\nu_{g,t}
&\leq C_g,
&& \forall g\in\mathcal G,\ t\in\mathcal T,
\label{eq:dual-generation}\\
-\lambda_{n,t}+\rho_{j,n,t}-\eta\gamma_{j,n,t}
&\leq \frac{d_j}{2},
&& \forall j\in\mathcal I,\ n\in\mathcal N,\ t\in\mathcal T,
\label{eq:dual-charge}\\
\lambda_{n,t}+\sigma_{j,n,t}+\eta^{-1}\gamma_{j,n,t}
&\leq \frac{d_j}{2},
&& \forall j\in\mathcal I,\ n\in\mathcal N,\ t\in\mathcal T,
\label{eq:dual-discharge}\\
-\lambda_{n,t}+\lambda_t^{\mathrm{sys}}
+\sum_{l\in\mathcal L}H_{l,n}
\left(\mu^+_{l,t}+\mu^-_{l,t}\right)
&=0,
&& \forall n\in\mathcal N,\ t\in\mathcal T.
\label{eq:dual-injection}
\end{align}
\end{subequations}

For compact state-of-charge stationarity, set
$\gamma_{j,n,0}=\gamma_{j,n,T+1}=0$. Then
\begin{subequations}\label{eq:dual-soc}
\begin{align}
A^E_{j,n,\tau}
:={}&
\delta_{j,n,\tau}
+\gamma_{j,n,\tau}
-\gamma_{j,n,\tau+1}\notag\\
&+
\left(\mathbf 1_{\{\tau=0\}}-\mathbf 1_{\{\tau=T\}}\right)
\rho^{\mathrm{per}}_{j,n},
\label{eq:soc-stationarity-expression}\\
A^E_{j,n,\tau}
&\leq 0,
&& \forall j\in\mathcal I,\ n\in\mathcal N,\ \tau\in\mathcal T_0.
\label{eq:soc-stationarity}
\end{align}
\end{subequations}

The corresponding dual objective is
\begin{align}
Z^D
:={}&
\sum_{n\in\mathcal N}\sum_{t\in\mathcal T}
D_{n,t}\lambda_{n,t}
+\sum_{g\in\mathcal G}\sum_{t\in\mathcal T}
\overline P_{g,t}\nu_{g,t}
\notag\\
&+\sum_{l\in\mathcal L}\sum_{t\in\mathcal T}
\overline F_l\left(\mu^+_{l,t}-\mu^-_{l,t}\right)
\notag\\
&+\sum_{j\in\mathcal I}\sum_{n\in\mathcal N}\sum_{t\in\mathcal T}
X^P_{j,n}\left(\rho_{j,n,t}+\sigma_{j,n,t}\right)
\notag\\
&+\sum_{j\in\mathcal I}\sum_{n\in\mathcal N}
\sum_{\tau\in\mathcal T_0}
X^E_{j,n}\delta_{j,n,\tau}.
\label{eq:dual-objective}
\end{align}
Exact lower-level optimality is imposed by
\begin{equation}\label{eq:strong-duality}
Z^P=Z^D.
\end{equation}
Together, primal feasibility \eqref{eq:llp-primal}, dual feasibility
\eqref{eq:llp-dual}--\eqref{eq:dual-soc}, and
\eqref{eq:strong-duality} are necessary and sufficient for optimality of the
lower-level LP. Explicit complementarity products are therefore unnecessary.

\section{Capacity-only EPEC}

Let $y$ and $u$ collect the lower-level primal and dual variables,
respectively. Given $X_{-i}$, investor $i$ solves the single-level MPEC
\begin{equation}\label{eq:best-response}
\begin{aligned}
\underset{X_i^P,X_i^E,y,u}{\operatorname{maximize}}
\quad & \Pi_i\\
\text{subject to}\quad
&\eqref{eq:investment},\
\eqref{eq:llp-primal},\
\eqref{eq:nonpositive-duals}\text{--}\eqref{eq:dual-soc},\
\eqref{eq:strong-duality},
\end{aligned}
\qquad \text{given }X_{-i}.
\end{equation}
Products between endogenous capacities and dual variables in
\eqref{eq:dual-objective}, together with LMP settlement in
\eqref{eq:profit}, make the best-response problem nonconvex.

Define $\operatorname{BR}_i(X_{-i})$ as investor $i$'s best-response
correspondence. A capacity profile $X^\star$ is an EPEC equilibrium if
\begin{equation}\label{eq:epec}
X_i^\star\in\operatorname{BR}_i(X_{-i}^\star),
\qquad \forall i\in\mathcal I.
\end{equation}
The implementation searches for a fixed point using simultaneous
Gauss--Jacobi updates. At sweep $k$, all best responses are computed against
the same frozen capacity profile:
\begin{subequations}\label{eq:jacobi}
\begin{align}
\widehat X_i^k
&\in\operatorname{BR}_i(X_{-i}^k),
&& \forall i\in\mathcal I,
\label{eq:raw-best-response}\\
X_i^{k+1}
&=(1-\omega)X_i^k+\omega\widehat X_i^k,
&& \forall i\in\mathcal I.
\label{eq:damped-update}
\end{align}
\end{subequations}
Damping affects the applied iterate but not the equilibrium condition.
Accordingly, convergence is assessed from the undamped residual
\begin{equation}\label{eq:raw-residual}
r^k
:=\max_{i\in\mathcal I}
\left\lVert\widehat X_i^k-X_i^k\right\rVert_\infty,
\end{equation}
with separate implementation tolerances for power and energy capacity.

Because a fixed-capacity LP can admit multiple dual-optimal LMP vectors, the
strong-duality formulation uses an optimistic convention: the embedded dual
vector may be selected in the active investor's favour. No secondary
minimum-norm or other price-selection rule is imposed. Consequently, embedded
investor-specific LMPs should be distinguished from prices obtained by an
independent common-market re-clear.

\section{Implementation}

The maintained implementation is organized as follows:
\begin{itemize}
\item \texttt{primal\_market\_clearing\_model.py}: lower-level primal LP;
\item \texttt{dual\_market\_clearing\_model.py}: explicit lower-level dual LP;
\item \texttt{mpec\_strong\_duality.py}: single-investor best-response MPEC;
\item \texttt{jacobi\_diagonalization.py}: simultaneous multi-investor updates;
\item \texttt{run\_model.py}: solver settings, convergence, output, and resume.
\end{itemize}
