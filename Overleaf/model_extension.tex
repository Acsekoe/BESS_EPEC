\section{Scope}

The model is a deterministic capacity-investment EPEC for battery energy
storage systems (BESSs) in a nodal electricity market. Investor $i$ chooses
only nodal power and energy capacity. For fixed capacities, the independent
system operator (ISO) clears generation and BESS operation at minimum physical
dispatch cost.

This clean formulation uses fixed demand, exact nodal and system balances, and
exact lower-level strong duality. It contains no strategic bid variables, load
shedding, dispatch regularization, leader-side price penalty, proximal term,
Tikhonov term, minimum-norm price selector, or relaxed complementarity
condition. Rival capacities are fixed within each best response and updated by
diagonalization outside the mathematical program.

\section{Notation}

Investors, nodes, hours, generators, and lines are indexed by
$i,j\in\mathcal I$, $n\in\mathcal N$, $t\in\mathcal T$, $g\in\mathcal G$,
and $l\in\mathcal L$. The set $\mathcal G_n$ contains the generators at node
$n$, and $\mathcal T_0=\{0,\ldots,T\}$ indexes state of charge.

The principal parameters are fixed demand $D_{n,t}$, generation cost $C_g$,
available generation $\overline P_{g,t}$, PTDF $H_{l,n}$, line limit
$\overline F_l$, one-way BESS efficiency $\eta$, nodal connection limit
$\overline X_n$, duration limits $r^{\min},r^{\max}$, degradation cost $d_j$,
investment costs $c_i^P,c_i^E$, WACC $w_i$, lifetime $L_i$, and investor $i$'s
ownership share $s_{i,g}$ of generator $g$.

Investor $i$ chooses $X^P_{i,n}$ and $X^E_{i,n}$. The ISO chooses generation
$P_{g,t}$, charging $p^{ch}_{j,n,t}$, discharging $p^{dis}_{j,n,t}$, stored
energy $e_{j,n,\tau}$, and net injection $z_{n,t}$. The nodal-balance dual
$\lambda_{n,t}$ is the locational marginal price (LMP).

\section{Investor problem}

Given rival capacities $X_{-i}$, investor $i$ satisfies
\begin{subequations}\label{eq:investment}
\begin{align}
0\le X^P_{i,n}
&\le \overline X_n-\sum_{j\ne i}X^P_{j,n},
&&\forall n,\label{eq:headroom}\\
r^{\min}X^P_{i,n}\le X^E_{i,n}
&\le r^{\max}X^P_{i,n},
&&\forall n.\label{eq:duration}
\end{align}
\end{subequations}
Its capital-recovery factor and daily investment cost are
\begin{align}
CRF_i&=\frac{w_i(1+w_i)^{L_i}}{(1+w_i)^{L_i}-1},\\
K_i&=\frac{CRF_i}{365.25}\sum_n
\left(c_i^P X^P_{i,n}+c_i^E X^E_{i,n}\right).
\end{align}

Daily profit is
\begin{subequations}\label{eq:profit}
\begin{align}
\Pi_i={}&R_i^{BESS}+R_i^{gen}-C_i^{deg}-K_i,\\
R_i^{BESS}={}&\sum_{n,t}\lambda_{n,t}
\left(p^{dis}_{i,n,t}-p^{ch}_{i,n,t}\right),\\
R_i^{gen}={}&\sum_{g,t}s_{i,g}
\left(\lambda_{n(g),t}-C_g\right)P_{g,t},\\
C_i^{deg}={}&\frac{d_i}{2}\sum_{n,t}
\left(p^{ch}_{i,n,t}+p^{dis}_{i,n,t}\right).
\end{align}
\end{subequations}
For a stand-alone merchant, $s_{i,g}=0$. A portfolio investor also internalizes
the spot rent of its owned generation shares.

\section{Clean lower-level market}

For fixed capacities $X=(X_i,X_{-i})$, the ISO solves
\begin{equation}\label{eq:llp-objective}
\min\quad Z^P=
\sum_{g,t}C_gP_{g,t}
+\frac12\sum_{j,n,t}d_j
\left(p^{ch}_{j,n,t}+p^{dis}_{j,n,t}\right)
\end{equation}
subject to
\begin{subequations}\label{eq:llp-primal}
\begin{align}
z_{n,t}
&=\sum_{g\in\mathcal G_n}P_{g,t}
+\sum_j\left(p^{dis}_{j,n,t}-p^{ch}_{j,n,t}\right)-D_{n,t},
&&\forall n,t,\label{eq:nodal-balance}\\
\sum_nz_{n,t}&=0,
&&\forall t,\label{eq:system-balance}\\
0\le P_{g,t}&\le\overline P_{g,t},
&&\forall g,t,\\
-\overline F_l\le\sum_nH_{l,n}z_{n,t}&\le\overline F_l,
&&\forall l,t,\\
0\le p^{ch}_{j,n,t}&\le X^P_{j,n},
&&\forall j,n,t,\\
0\le p^{dis}_{j,n,t}&\le X^P_{j,n},
&&\forall j,n,t,\\
e_{j,n,t}&=e_{j,n,t-1}+\eta p^{ch}_{j,n,t}
-\eta^{-1}p^{dis}_{j,n,t},
&&\forall j,n,t,\\
0\le e_{j,n,\tau}&\le X^E_{j,n},
&&\forall j,n,\tau,\\
e_{j,n,0}&=e_{j,n,T},
&&\forall j,n.
\end{align}
\end{subequations}
The daily cycle is periodic; no fixed initial state of charge is imposed.

\section{Exact lower-level optimality}

The lower-level linear program is embedded through primal feasibility, dual
feasibility, and strong duality. Let $\lambda_{n,t}$ and $\lambda_t^{sys}$ be
the free multipliers of \eqref{eq:nodal-balance} and
\eqref{eq:system-balance}. Let
\begin{equation}
\nu_{g,t},\mu^{up}_{l,t},\rho_{j,n,t},\sigma_{j,n,t},
\delta_{j,n,\tau}\le0,
\qquad
\mu^{dn}_{l,t}\ge0,
\end{equation}
and let $\gamma_{j,n,t}$ and $\rho^{per}_{j,n}$ be free. Dual feasibility is
\begin{subequations}\label{eq:llp-dual}
\begin{align}
\lambda_{n(g),t}+\nu_{g,t}&\le C_g,
&&\forall g,t,\\
-\lambda_{n,t}+\rho_{j,n,t}-\eta\gamma_{j,n,t}
&\le\frac{d_j}{2},
&&\forall j,n,t,\\
\lambda_{n,t}+\sigma_{j,n,t}+\eta^{-1}\gamma_{j,n,t}
&\le\frac{d_j}{2},
&&\forall j,n,t,\\
-\lambda_{n,t}+\lambda_t^{sys}
+\sum_lH_{l,n}\left(\mu^{up}_{l,t}+\mu^{dn}_{l,t}\right)&=0,
&&\forall n,t.
\end{align}
\end{subequations}
For stored energy, define
\begin{align}
A^E_{j,n,\tau}:={}&\delta_{j,n,\tau}
+\mathbf 1_{\{\tau\in\mathcal T\}}\gamma_{j,n,\tau}
-\mathbf 1_{\{\tau+1\in\mathcal T\}}\gamma_{j,n,\tau+1}\notag\\
&+\mathbf 1_{\{\tau=0\}}\rho^{per}_{j,n}
-\mathbf 1_{\{\tau=T\}}\rho^{per}_{j,n},
\end{align}
with
\begin{equation}
A^E_{j,n,\tau}\le0,
\qquad\forall j,n,\tau.
\end{equation}

The dual objective is
\begin{align}
Z^D={}&
\sum_{n,t}D_{n,t}\lambda_{n,t}
+\sum_{g,t}\overline P_{g,t}\nu_{g,t}
+\sum_{l,t}\overline F_l
\left(\mu^{up}_{l,t}-\mu^{dn}_{l,t}\right)\notag\\
&+\sum_{j,n,t}X^P_{j,n}
\left(\rho_{j,n,t}+\sigma_{j,n,t}\right)
+\sum_{j,n,\tau}X^E_{j,n}\delta_{j,n,\tau}.
\end{align}
The exact lower-level optimality condition is
\begin{equation}\label{eq:strong-duality}
Z^P=Z^D.
\end{equation}
Together with \eqref{eq:llp-primal} and \eqref{eq:llp-dual},
\eqref{eq:strong-duality} is necessary and sufficient for an optimal solution
of the clean lower-level LP. Explicit complementarity products are unnecessary.

\section{Capacity-only EPEC}

Investor $i$'s best response is the single-level MPEC
\begin{equation}\label{eq:best-response}
\boxed{
\begin{aligned}
\max_{X_i^P,X_i^E,\,y,\,u}\quad &\Pi_i\\
\text{s.t.}\quad
&\eqref{eq:investment},\quad
\eqref{eq:llp-primal},\quad
\eqref{eq:llp-dual},\quad
\eqref{eq:strong-duality},
\end{aligned}}
\qquad\text{given }X_{-i},
\end{equation}
where $y$ collects the lower-level primal variables and $u$ its dual variables.
Products between capacities and dual variables, together with LMP settlement in
\eqref{eq:profit}, make the best-response problem nonconvex.

A capacity profile $X^*$ is an EPEC equilibrium if
\begin{equation}\label{eq:epec}
X_i^*\in BR_i(X_{-i}^*),
\qquad\forall i\in\mathcal I.
\end{equation}
The implementation searches for this fixed point with Gauss--Jacobi or
Gauss--Seidel diagonalization. With Jacobi damping $0<\omega\le1$,
\begin{equation}
X_i^{k+1}=(1-\omega)X_i^k+\omega\widehat X_i^k,
\qquad
\widehat X_i^k\in BR_i(X_{-i}^k).
\end{equation}
Damping changes only the numerical fixed-point iteration. Formal convergence
must be assessed using the undamped best-response residuals.

Because a fixed-capacity LP can admit multiple dual-optimal LMP vectors, this
clean strong-duality formulation uses an optimistic convention: the embedded
dual vector may be selected in the investor's favour. No secondary price rule
is imposed. After a run, final capacities are independently re-cleared to report
common physical dispatch and reference-market prices; disagreement between
embedded and reference LMPs must be disclosed.
